# Production configuration for this Next.js service.
# Create the protected copy with configure_settings.py (see README), which asks
# for or generates every CHANGE_ME{NAME} input.
# NEXT_PUBLIC_* values are embedded in the browser bundle during image build;
# all other values are server/container settings and must not be exposed.

### Unprivileged Node.js runtime
# Source path used while assembling the image. Example: /app
REPO_PATH='/app'
# Final application path inside the image. Example: /app
INSTALL_PATH='/app'
# Numeric UID/GID for the unprivileged Node process and writable Next.js cache.
# Example: 1000. Use IDs compatible with the deployment host policy.
APP_UID='1000'

# Internal port used by `next start`, health checks, and Apache. Example: 3000
APP_PORT='3000'

### Public browser build configuration (never put secrets in NEXT_PUBLIC_* values)

# Next.js proxies API requests. Example: /api/v1
NEXT_PUBLIC_API_BASE_URL='/api/v1'
# Public Keycloak origin reachable by users' browsers, without a realm path.
# Example: https://auth.pathocore.example.org

# Keycloak realm containing the frontend client. Example: pathocore
NEXT_PUBLIC_KEYCLOAK_REALM='CHANGE_ME{KEYCLOAK_REALM}'
# Public Keycloak client ID registered for this frontend. Example: pathocore-web
NEXT_PUBLIC_KEYCLOAK_CLIENT_ID='CHANGE_ME{KEYCLOAK_FRONTEND_CLIENT_ID}'

### Server-only runtime configuration
# Server-side proxy targets are not configured here. Compose generates
# <UPSTREAM>_PROXY_TARGET=http://<upstream>:<upstream APP_PORT> for every
# service listed in SERVICES.<this service>.UPSTREAMS.
# Secret used to encrypt/sign authentication state. Generate independently for
# every deployment, for example with: openssl rand -base64 32
AUTH_SECRET='CHANGE_ME{AUTH_SECRET}'
# Canonical public application URL used by Auth.js callbacks, including scheme
# and no trailing slash. Example: https://pathocore.example.org
AUTH_URL='https://CHANGE_ME{PUBLIC_HOSTNAME}'
